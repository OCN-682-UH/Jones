## Filter data for country by 2020
plastics_2020 <- plastics |>
  filter(year == 2020) |>
  group_by(country) |>
  summarize(
    grand_total = sum(grand_total, na.rm = TRUE),
    .groups = "drop"
  )

view(plastics_2020)

## Found country names that are not aligned were USA, UK, and Korea, so I need to mutate those
plastics_tojoin_2020 <- plastics_2020 |> mutate(
  country = recode(country, "United States of America" = "USA", "United Kingdom of Great Britain & Northern Ireland" = "UK", "Korea" = "South Korea")
)

## Now, I will re-join the datasets
world_plastics <- world |>
  left_join(
    plastics_tojoin_2020,
    by = c("region" = "country")
  )
view(world_plastics)

## Remove the NAs from the dataset
wp_clean <- world_plastics |>
  drop_na()
view(wp_clean)

# Make a map of the world with plastic data overlaid
ggplot()+
  geom_polygon(data = world, aes(
    x = long, y = lat, group = group),
    fill = "grey80",color = "black", linewidth = 0.1) +
  geom_polygon(data = wp_clean,
               aes(x = long, y = lat,
                   group = group,
                   fill = grand_total),
               color = "black",
               linewidth = 0.1) +
  scale_fill_viridis_c(
    option = "mako",
    na.value = "grey80",
    name = "Total Plastics"
  ) +
  coord_fixed(1.3) +
  theme_void() +
  labs(
    title = "Plastic Waste Collected per Country in 2020",
    subtitle = "From the Break Free From Plastic Global Cleanup Data",
    caption = "Grey countries have no matching data"
  )
